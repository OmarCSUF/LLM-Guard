\begin{algorithm}[h]
\caption{Layer-1 detection with input views}
\label{alg:l1}
\begin{algorithmic}[1]
\REQUIRE prompt $x$; stacker $F$; scaler $T^\star$; thresholds
         $\tau_{\mathrm{low}},\tau_{\mathrm{high}}$; flags \textsc{norm}, \textsc{views}
\ENSURE routed score $s^\star$ and decision $d$
\IF{$x$ is empty}
  \RETURN $(0.5,\ \textsc{gray zone})$
\ENDIF
\STATE $s^\star \leftarrow \textsc{Score}(x)$
\STATE $\mathcal{F} \leftarrow \emptyset$
\IF{\textsc{norm}}
  \STATE $(x_{\mathrm{rec}},A) \leftarrow \textsc{Normalise}(x)$
  \IF{$A \neq \emptyset$}
    \STATE $\mathcal{F} \leftarrow \mathcal{F} \cup \{x_{\mathrm{rec}}\}$
  \ENDIF
  \STATE $\mathcal{F} \leftarrow \mathcal{F} \cup \textsc{Payloads}(x)$
         \COMMENT{each decoded blob alone}
\ENDIF
\IF{\textsc{views}}
  \STATE $\mathcal{F} \leftarrow \mathcal{F} \cup \textsc{ClauseViews}(x)$
\ENDIF
\FORALL{$x' \in \mathcal{F}$}
  \STATE $s^\star \leftarrow \max\bigl(s^\star,\ \textsc{Score}(x')\bigr)$
         \COMMENT{maximum only; cannot decrease}
\ENDFOR
\IF{$s^\star < \tau_{\mathrm{low}}$}
  \STATE $d \leftarrow \textsc{allow}$
\ELSIF{$s^\star > \tau_{\mathrm{high}}$}
  \STATE $d \leftarrow \textsc{mitigate}$
\ELSE
  \STATE $d \leftarrow \textsc{gray zone}$
\ENDIF
\RETURN $(s^\star,\ d)$
\STATE \rule{\linewidth}{0.3pt}
\STATE \textbf{procedure} \textsc{Score}$(t)$
\IF{$F$ reads the scoring-LM features}
  \STATE $f_{\mathrm{curv}} \leftarrow \textsc{Curvature}(t)$;\ \
         $f_{\mathrm{rank}} \leftarrow \textsc{LogRank}(t)$
\ELSE
  \STATE $f_{\mathrm{curv}},f_{\mathrm{rank}} \leftarrow 0.5,\,0.5$
         \COMMENT{skipped, not computed}
\ENDIF
\STATE $g \leftarrow \textsc{Classifier}(t)$ \COMMENT{$P(\mathrm{allow}\mid t)$}
\STATE $s_{\mathrm{raw}} \leftarrow
        1 - F.P_{\mathrm{allow}}(f_{\mathrm{curv}},f_{\mathrm{rank}},g)$
\RETURN $\sigma\bigl(\mathrm{logit}(s_{\mathrm{raw}})/T^\star\bigr)$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[h]
\caption{Layer-3 mitigation with the refusal veto}
\label{alg:l3}
\begin{algorithmic}[1]
\REQUIRE prompt $x$; category $c$; severity $v$; escalation flag $e$;
         constraint engine $E$; Guard-LLM $Q$; refuse-always set $\mathcal{R}$
\ENSURE the mode actually served, and its output
\STATE $\mathrm{trig} \leftarrow E.\textsc{Categories}(x)$
\IF{$c \in \mathcal{R}$ \OR $\mathrm{trig} \cap \mathcal{R} \neq \emptyset$}
  \STATE $m \leftarrow \textsc{refuse}$
         \COMMENT{$\mathrm{trig}$ supplies the category the L1 route lacks}
\ELSIF{$v \geq 0.99$ \OR $e$}
  \STATE $m \leftarrow \textsc{refuse}$
\ELSE
  \STATE $m \leftarrow$ configured default, which is \textsc{refuse}
\ENDIF
\IF{$m \neq \textsc{safe-rewrite}$}
  \RETURN $\textsc{Serve}(m)$
\ENDIF
\STATE $\mathrm{declined} \leftarrow \textsc{false}$;\ \ $C \leftarrow \emptyset$
\FOR{$k = 1$ \TO $K'$}
  \STATE $y \leftarrow Q.\textsc{Generate}(x,\ \text{temperature}=0.7)$
  \IF{$y$ is the sentinel \textsc{refuse} \OR $y$ reads as a prose refusal}
    \STATE $\mathrm{declined} \leftarrow \textsc{true}$
    \STATE \textbf{if} refuse-on-decline \textbf{then break}
           \COMMENT{the refusal veto}
  \ELSE
    \STATE $C \leftarrow C \cup \bigl\{(\,|E.\textsc{Check}(y)|,\ y)\bigr\}$
  \ENDIF
\ENDFOR
\IF{$\mathrm{declined}$ \AND refuse-on-decline}
  \RETURN $\textsc{Serve}(\textsc{refuse})$
\ENDIF
\STATE sort $C$ by violation count, ascending
\IF{$C \neq \emptyset$ \AND $C[0]$ has zero violations}
  \RETURN $\textsc{Serve}(\textsc{safe-rewrite},\ C[0])$
\ENDIF
\RETURN $\textsc{Serve}(\textsc{refuse})$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[h]
\caption{Cascade decision}
\label{alg:cascade}
\begin{algorithmic}[1]
\REQUIRE prompt $x$; detector $D$; judge $J$ or $\bot$; mitigator $G$ or $\bot$
\ENSURE prediction $\hat{y}$, harm score $h$, escalation flag
\STATE $(s^\star,d) \leftarrow D.\textsc{Detect}(x)$
       \COMMENT{Algorithm~\ref{alg:l1}}
\IF{$d = \textsc{allow}$}
  \RETURN $(1,\ s^\star,\ \textsc{false})$
\ENDIF
\IF{$d = \textsc{mitigate}$}
  \STATE \textbf{if} $G \neq \bot$ \textbf{then} $G.\textsc{FromL1}(x,s^\star)$
  \RETURN $(0,\ s^\star,\ \textsc{false})$
\ENDIF
\IF{$J = \bot$}
  \RETURN $(1,\ s^\star,\ \textsc{false})$
         \COMMENT{unresolved band allows}
\ENDIF
\STATE $r \leftarrow J.\textsc{Judge}(x)$
\STATE $h \leftarrow \tfrac{1}{2}\bigl(s^\star + r.\mathrm{severity}\bigr)$
\IF{$r.\mathrm{escalated}$}
  \STATE \textbf{if} $G \neq \bot$ \textbf{then} $G.\textsc{FromL2}(x,r)$
  \RETURN $(0,\ h,\ \textsc{true})$ \COMMENT{uncertainty blocks}
\ENDIF
\IF{$r.\mathrm{decision} = \textsc{allow}$}
  \RETURN $(1,\ h,\ \textsc{false})$
\ENDIF
\STATE \textbf{if} $G \neq \bot$ \textbf{then} $G.\textsc{FromL2}(x,r)$
\RETURN $(0,\ h,\ \textsc{false})$
\end{algorithmic}
\end{algorithm}

\section*{S1. Contents of the reproducibility package}

Every figure and table above is regenerated from stored per-prompt predictions
by the analysis scripts in the accompanying repository, released at
\url{https://github.com/OmarCSUF/LLM-Guard}; results files carry the
command, git commit, seed, package versions and a configuration snapshot.
The package contains, specifically: (i)~the pre-registration, the register of
configuration decisions taken on development data, the Layer-3 leak codebook,
the learner-study pre-registration and the five dated amendments, which record
the initially refused baseline, every post hoc arm and analysis, and the
withdrawn scoring run; (ii)~the freeze record, with SHA-256 digests of the
configuration, of 63 source files, of the Layer-1 artifacts of all four
checkpoints (LoRA adapters and stackers, about 9\,MB each, included), of the
three pool splits and of the thirteen benchmark and learner input files,
together with the git commit; (iii)~the final test set manifest, which records
the screening counts at every stage, the pinned revision of each source

\section*{S2. Metrics and statistical analysis}

Every arm returns a binary decision $\hat{y}_i$ and a scalar harm score per
prompt. The two operational rates reported throughout the article are
\begin{align}
\mathrm{safe\_blocked}   &= \frac{\bigl|\{i: y_i{=}1 \wedge \hat{y}_i{=}0\}\bigr|}
                                 {\bigl|\{i: y_i{=}1\}\bigr|}, \label{eq:sb}\\[2pt]
\mathrm{harmful\_allowed} &= \frac{\bigl|\{i: y_i{=}0 \wedge \hat{y}_i{=}1\}\bigr|}
                                 {\bigl|\{i: y_i{=}0\}\bigr|}, \label{eq:ha}
\end{align}
with escalated prompts counted as blocked because the configured mitigation is
refusal.

For a given source and pair of arms, the difference in rate is resampled by
drawing prompts with replacement within that source, applying the same index
set to both arms and taking the 2.5th and 97.5th percentiles of $B=2000$
replicates; every source is single-class, so this holds class composition
fixed. Differences in the pooled AUROC are resampled the same way, both AUROCs
recomputed on one prompt resample, so a small gap is read against an interval
rather than as an ordering.

The adversarial suite is summarised by the bypass rate over 3{,}600 variants,
with an interval from a cluster bootstrap that resamples the 200 seed requests,
since variants of one request are not independent. The learner study reports
the mean difference in blocking between original and corrected sentences, with
percentile intervals from bootstraps grouped by sentence, by essay and by
writer; the writer-grouped upper bound is compared with the pre-specified
two-point margin.

All bootstraps use $B=2000$ and $\alpha=0.05$ from a fixed study seed, each
comparison drawing from its own stream derived from that seed and the
comparison's name. Layer-1 scores are quantised to four decimals before any
threshold is applied, as the router serves them.

\section*{S3. Implementation and compute}

All runs used one NVIDIA GeForce RTX 4050 Laptop GPU (6\,GB, 5.64\,GiB
addressable) with CUDA 13.0, under Python 3.10.12 with
PyTorch 2.14.0, Transformers 4.57.6, bitsandbytes 0.50.2, PEFT 0.20.0,
Accelerate 1.15.0, scikit-learn 1.7.2 and NumPy 2.2.6. Layer~1 runs
unquantised. Generation is greedy at temperature 0 wherever a verdict is
decoded, so scoring is deterministic up to the numerical non-determinism of the
quantised kernels: re-scoring 60 prompts with Granite Guardian moved its harm
score by at most $6\times10^{-3}$ and changed no decision. Latency is
wall-clock time per prompt inside the scoring loop, excluding model loading, on
the same GPU for every arm: the median over the final test set, except for the
two Layer-1-dominated arms, whose runs recorded only the loop total and report
the mean. Every table and figure in
Section~\ref{sec:results} is regenerated from stored per-prompt predictions.

\section*{S4. Arms}

Prompts are truncated to 2{,}048 tokens before templating and verdicts decoded
greedily with at most eight new tokens, two for ShieldGemma. The guard
revisions are pinned: \texttt{81145486} for Granite Guardian,
\texttt{acf7aafa} for Llama Guard and \texttt{d1dffc9c} for ShieldGemma.
ShieldGemma's four policies are queried at four forward passes per prompt, and
every per-policy score is kept. The harm score of a guard is renormalised over
the first-token spellings of the two answers at the position where the verdict
is emitted. The 4-bit NF4 quantisation is forced by the hardware: the 6\,GB GPU
does not hold a 2\,B model in \texttt{bfloat16} beside its activations.

No primary publication describes Llama Guard 3 1B itself, and the article cites
none for it: the model is distributed through its model card, and the only
publication we could verify that addresses the 1\,B model evaluates a
compressed INT4 derivative against it.

For the two post hoc substitute-judge arms, the prompt version was chosen on
pool validation by the criterion that chose the incumbent's.

\section*{S5. Training pool}

Layer~1 is fine-tuned as \texttt{roberta-base} with LoRA rank $8$,
$\alpha=16$, dropout $0.1$, 512-token inputs, batch size 8, learning rate
$2\times10^{-4}$ and at most five epochs with early stopping after two without
improvement. The pool's near-duplicate rule uses 64 permutations and 16 bands
with seed 42, and cross-split overlap is re-checked with independent hash
permutations. The judge's prompt version was chosen on 300 pool validation
prompts, half per class, with seed 42.

\section*{S6. Development benchmarks}

The five development benchmarks break down as follows. JailbreakBench
contributes 200 prompts, 100 per class; XSTest 450, of which 250 are safe and
200 unsafe; HarmBench 200, all harmful; the ToxicChat 0124 test split 5{,}083
rows loaded, 132 exact duplicates removed and 4{,}951 kept, of which 4{,}597
are safe and 354 harmful; and OR-Bench 1{,}974, of which 1{,}319 are safe and
655 harmful.

\section*{S7. The final test set}

The screening steps of Section~\ref{sec:finalset} attribute to the individual

\section*{S9. Grammatical-form sensitivity by CEFR level and by seed}

By CEFR level the writer-grouped differences are $-0.77$ $[-2.06, +0.49]$ (A),
$-1.61$ $[-2.88, -0.42]$ (B) and $-1.26$ $[-2.49, -0.08]$ (C), each with its
upper bound inside the pre-specified two-point non-inferiority margin. The
pre-registered replication on the three repeat-study checkpoints gives pooled
differences of $-2.05$ $[-2.86, -1.24]$, $+0.22$ $[-0.53, +0.97]$ and $-0.68$
$[-1.37, +0.05]$ for seeds 42, 43 and 44, all within the margin at the pooled
level; one level cell, A under seed 43, exceeds it at $+1.68$ $[+0.29, +2.97]$.
The claim therefore holds on every checkpoint pooled over levels and fails in
one of twelve level cells.

\section*{S10. Extended discussion}

Three bounds apply to the word ``independent''. Independence from the authors'
decisions is not independence from the models' pre-training: the seven sources
are public, and the near-duplicate screen addresses what the authors saw, not
what the models saw. ShieldGemma, the two Llama judge arms and the ablation of
Section~\ref{sec:ablation} were added once the pre-registered results were
known, so the prompts are as independent for them as for any arm but the arms
may not be, and the paper marks them rather than relying on them. And every
source is single-class, so pooled AUROC partly measures source separability:
per-source rates are the primary evidence, and pooled ranking is read only
against its paired intervals.

Widening the routing band to $0.05/0.95$ recovers about two of the \num{3.7}
points between the cascade's harmful-allowed rate and the judge's at five times
the judge traffic (Fig.~\ref{fig:frontiers}): a partial remedy only.

Beyond its interval the count of bypasses is uncertain in three ways that run in
the same direction: variants of one request are not independent, so the
effective sample is nearer the 200 requests than the 3{,}600 variants and
$[2.5, 6.8]$ is wide relative to its estimate; the suite is development data, so
the rate is optimistic for exactly the families the view targets; and two
families came from the judge's own model family, limiting attack diversity.

\textbf{The scoring-LM signals carry no usable information.} They changed almost
no decisions in the ablation yet cost \SI{32}{\milli\second} of Layer~1's
\SI{38}{\milli\second}, and harmful prompts do not separate from benign ones
along the axis perturbation curvature and log-rank statistics were designed for.
The rejection was taken on the development benchmarks, so its benefit on fresh
data is not established; its cost is, and that suffices.

\textbf{The input-view mechanisms are inert on plain prompts.} Normalisation
changes no decision on any of the 3{,}453 prompts on any of the four
checkpoints, and the leading-clause view changes between seven and nine, every
one a safe prompt blocked; the only evidence that either helps is the
adversarial suite that shaped them, and they are cheap to keep.

The two defects were these. A temperature fitted against the unflipped label
referred every prompt without raising an error, and the withdrawn guard run read
a harm score at the wrong generation step and ranked inconsistently with its own
decisions; both were caught by an internal inconsistency a reader could have
missed rather than by the software. The invariants now asserted include the sign
of the score against the label and the probability mass at the step being read.

A margin smaller than the spread cannot be attributed to the intervention rather
than the draw, which is why the Llama Guard and ShieldGemma margins are not
treated as findings even where a paired interval excludes zero: that bootstrap
varies prompts for a fixed pair of models, not retrainings of one of them. The
configured checkpoint is the most favourable of the four on pooled ranking and
CatQA and was the one retained during development, and whether the development
benchmarks favoured it cannot now be determined, so the seed rows of
Table~\ref{tab:seeds} report the recipe's typical performance rather than its
worst. And the two checkpoints trained with seed 42 differ by \num{0.060}, the
entire spread, so the variation is not attributable to the seed value alone;
those runs used different scripts, and whatever else differs between them
contributes. The study does not separate these sources, so ``seed sensitivity''
in Section~\ref{sec:seeds} means run-to-run variation of the recipe --- a limit
of the repeat study's design, not of its result, since how far the recipe moves
every reported quantity is what a reader needs.

The limits of the learner study matter as much. It varies form within a writer,
not writers, groups or any other attribute, and uses no demographic information,
so it is evidence about form sensitivity, not about fairness across
populations. Both arms block about a fifth of the sentences, and because every
sentence carries the safe label without independent review, high over-refusal on
essay prose cannot be distinguished from label error. The replication across the
three repeat-study checkpoints holds at the pooled level, but the one level cell
that exceeds the margin, and the pooled difference that changes sign under seed
43, show that about one point is within what retraining moves: direction as well
as size should be read against that spread.

The episodes behind the evaluation recommendations were these. An end-to-end
number hid a middle layer used on one prompt in thirty-five; two of this paper's
three external margins fall inside the retraining spread; the study accreted
three post hoc arms and a sixteen-arm ablation once its results were known, each
for a defensible reason, and only the marks let a reader weight them; and two
defects here produced numbers that were plausible and wrong and raised no error.

\section*{S11. Recovered: related work}

The following characterisations of the cited work were carried by the article
before it was compressed to the page limit. They are restored here because the
article now names each work without describing it.

Production content moderation predates instruction-tuned language models.
Markov \emph{et al.} describe a deployed classifier built around a designed
content taxonomy and labelling instructions, data quality control, and an
active-learning pipeline for the rare events a static corpus under-represents.
The current generation instead fine-tunes an instruction-following language
model to emit the verdict. Llama Guard frames moderation as classification over
a hazard taxonomy supplied in the prompt, so that categories can be revised
without retraining; its third version ships alongside the Llama~3 models, and a
compressed INT4 derivative of the 1\,B model, evaluated against it, is described
separately.

A different line keeps the cost low by reading the generating model's own
intermediate representations rather than running a separate guard: multi-layer
latent prototypes classify a prompt from hidden states at negligible added cost
and can be fitted to any model.

Spending computation in proportion to difficulty is an old construction. The
boosted cascade of Viola and Jones rejects the easy majority of candidate
windows with a few cheap features and reserves expensive classifiers for what
survives; the reject option of selective classification formalises the same
trade, exchanging coverage for accuracy on the inputs a model chooses to answer.
FrugalGPT applies the construction to language models directly, chaining cheaper
to costlier models and learning the thresholds at which to escalate. The
selective-classification literature also offers constructions in which the
deferral rule is learned jointly with the classifier rather than thresholded on
a post hoc calibrated score, and cascaded human--AI decision frameworks learn
the deferral and abstention policies and adapt them online from feedback.

\section*{S12. Recovered: the forced-prefix reading}

Two details of the Layer-2 reading matter for the judge-swap arm of
Section~\ref{sec:judgeswap}. First, the sets $\mathcal{A}$ and $\mathcal{M}$ are
the first-token identifiers of the four spellings of each word, with and without
a leading space and with and without capitalisation, after removing any
identifier the two words share, so the reading is defined for any tokenizer in
which the two words start differently.

\section*{S13. Recovered: the guards' own scoring rules}

Each guard's harm score is read with that vendor's own scoring rule. Llama
Guard's authors read the classifier score from the probability of the first
token; Granite Guardian's normalise the generation probabilities of the safe and
unsafe tokens, aggregated over their spellings, by a softmax across the two; and
ShieldGemma's model card computes the score as the softmax probability of
\texttt{Yes} over the \texttt{Yes} and \texttt{No} logits, the scoring mode its
paper directs users to.

Native verdicts are operating points chosen by each vendor for its own
deployment, and comparing them compares those choices as much as the models.
This is why Section~\ref{sec:heldout} reports the matched reading beside the
native one.
